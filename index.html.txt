        // Estado del Carrito
        let cart = [];
        let shippingCost = 0.00;

        function calculateItemPrice(basePrice, qty) {
            let discount = 0;
            if (qty >= 12) {
                discount = 0.20; // 20% descuento mayoreo
            } else if (qty >= 6) {
                discount = 0.10; // 10% descuento mayoreo
            }
            const unitFinalPrice = basePrice * (1 - discount);
            return unitFinalPrice * qty;
        }

        function addToCart(type) {
            let sizeSelect, qtyInput, namePrefix, basePriceKey;
            
            if (type === 'lavender') {
                sizeSelect = document.getElementById('lavender-size');
                qtyInput = document.getElementById('lavender-qty');
                namePrefix = 'Jabón de Lavanda';
            } else if (type === 'carbon') {
                sizeSelect = document.getElementById('carbon-size');
                qtyInput = document.getElementById('carbon-qty');
                namePrefix = 'Jabón de Carbón Activado';
            } else if (type === 'rice') {
                sizeSelect = document.getElementById('rice-size');
                qtyInput = document.getElementById('rice-qty');
                namePrefix = 'Jabón de Arroz Blanco';
            }

            const size = sizeSelect.value;
            const basePrice = Number(sizeSelect.options[sizeSelect.selectedIndex].getAttribute('data-price'));
            const qty = parseInt(qtyInput.value);

            if (qty <= 0) return alert('Ingresa una cantidad válida');

            const totalItemPrice = calculateItemPrice(basePrice, qty);

            // Buscar si ya existe en el carrito
            const existingIndex = cart.findIndex(item => item.type === type && item.size === size);
            if (existingIndex > -1) {
                cart[existingIndex].qty += qty;
                cart[existingIndex].totalPrice = calculateItemPrice(basePrice, cart[existingIndex].qty);
            } else {
                cart.push({
                    type: type,
                    name: `${namePrefix} (${size.toUpperCase()})`,
                    basePrice: basePrice,
                    qty: qty,
                    totalPrice: totalItemPrice
                });
            }

            updateCartUI();
            alert('¡Producto agregado al carrito exitosamente!');
        }

        function updateCartUI() {
            let totalCount = 0;
            let subtotal = 0;

            cart.forEach(item => {
                totalCount += item.qty;
                subtotal += item.totalPrice;
            });

            const iva = subtotal * 0.16;
            const totalFinal = subtotal + iva + shippingCost;

            document.getElementById('cart-count').innerText = totalCount;
            document.getElementById('cart-total').innerText = totalFinal.toFixed(2);

            document.getElementById('summary-subtotal').innerText = `$${subtotal.toFixed(2)} MXN`;
            document.getElementById('summary-iva').innerText = `$${iva.toFixed(2)} MXN`;
            document.getElementById('summary-shipping').innerText = `$${shippingCost.toFixed(2)} MXN`;
            document.getElementById('summary-total-final').innerText = `$${totalFinal.toFixed(2)} MXN`;

            convertCurrency();
        }

        // Mapa Interactivo y Cálculo de Envío por Distancia Redondo
        const storeCoords = [25.6714, -100.3099]; // Taller Base (Monterrey/Zona Metropolitana)
        const map = L.map('map').setView(storeCoords, 13);
        L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {

        let marker = null;

            const latlng = e.latlng;
            if (marker) {
                marker.setLatLng(latlng);
            } else {
                marker = L.marker(latlng).addTo(map);
            }

            // Calcular distancia en km usando Haversine simple
            const distanceKm = getDistanceFromLatLonInKm(storeCoords[0], storeCoords[1], latlng.lat, latlng.lng);
            
            // Regla de envío: $15 por km, precio redondo (Math.round)
            let rawShipping = distanceKm * 15;
            if (rawShipping < 40) rawShipping = 40; // Tarifa mínima de envío
            shippingCost = Math.round(rawShipping); // Precio redondo requerido

            document.getElementById('input-address').value = `Lat: ${latlng.lat.toFixed(4)}, Lng: ${latlng.lng.toFixed(4)} (${distanceKm.toFixed(1)} km)`;
            
            updateCartUI();
        });

        function getDistanceFromLatLonInKm(lat1, lon1, lat2, lon2) {
            const R = 6371; // Radio de la tierra en km
            const dLat = deg2rad(lat2-lat1);
            const dLon = deg2rad(lon2-lon1);
            const a = 
                Math.sin(dLat/2) * Math.sin(dLat/2) +
                Math.cos(deg2rad(lat1)) * Math.cos(deg2rad(lat2)) * 
                Math.sin(dLon/2) * Math.sin(dLon/2); 
            const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1-a)); 
            return R * c;
        }

        function deg2rad(deg) {
            return deg * (Math.PI/180);
        }

        async function fetchWeatherSuggestion() {
            try {
                // Coordenadas fijas para ejemplo (Monterrey)
                const res = await fetch('https://api.open-meteo.com/v1/forecast?latitude=25.67&longitude=-100.31&current_weather=true');
                const data = await res.json();
                const temp = data.current_weather.temperature;
                
                let suggestion = `Temperatura actual: ${temp}°C. `;
                if (temp > 28) {
                    suggestion += "¡Día caluroso! Te sugerimos altamente el **Jabón de Carbón Activado** para controlar la grasa y refrescar tu piel a profundidad.";
                } else {
                    suggestion += "Clima fresco o templado. El **Jabón Artesanal de Lavanda** es perfecto para relajar tu piel e hidratarla.";
                }
                document.getElementById('weather-suggestion').innerHTML = suggestion;
            } catch (error) {
                document.getElementById('weather-suggestion').innerText = "Clima no disponible temporalmente. Te recomendamos probar nuestra Lavanda relajante.";
            }
        }
        fetchWeatherSuggestion();

        // Tecnología Externa 2: Tasas de Cambio de Divisas en Tiempo Real (API Abierta)
        let exchangeRates = { USD: 0.058, EUR: 0.053 }; // Valores por defecto de respaldo
        async function fetchExchangeRates() {
            try {
                const res = await fetch('https://open.er-api.com/v6/latest/MXN');
                const data = await res.json();
                if(data && data.rates) {
                    exchangeRates.USD = data.rates.USD;
                    exchangeRates.EUR = data.rates.EUR;
                }
            } catch (e) {
                console.log("Usando tasas estándar de respaldo");
            }
            convertCurrency();
        }

        function convertCurrency() {
            const currency = document.getElementById('currency-select').value;
            let subtotal = 0;
            cart.forEach(item => subtotal += item.totalPrice);
            const iva = subtotal * 0.16;
            const totalFinal = subtotal + iva + shippingCost;

            const rate = exchangeRates[currency] || 0.055;
            const converted = totalFinal * rate;

            document.getElementById('converted-total').innerText = `Total convertido: $${converted.toFixed(2)} ${currency}`;
        }
        fetchExchangeRates();

        // Procesar Compra y Emitir Ticket
        function processCheckout(event) {
            event.preventDefault();
            if (cart.length === 0) {
                alert('Tu carrito está vacío. Agrega al menos un jabón para continuar.');
                return;
            }
            if (shippingCost === 0 && marker === null) {
                alert('Por favor selecciona tu ubicación de entrega en el mapa para calcular el costo de envío.');
                return;
            }

            let subtotal = 0;
            let ticketHtml = `<p><strong>Fecha:</strong> ${new Date().toLocaleString()}</p>`;
            ticketHtml += `<p><strong>Cliente:</strong> Ficticio / Verificado con Tarjeta</p><br>`;
            ticketHtml += `<p class="font-bold underline">Detalle de Productos:</p>`;

            cart.forEach(item => {
                subtotal += item.totalPrice;
                ticketHtml += `<div class="flex justify-between"><span>${item.qty}x ${item.name}</span> <span>$${item.totalPrice.toFixed(2)} MXN</span></div>`;
            });

            const iva = subtotal * 0.16;
            const finalTotal = subtotal + iva + shippingCost;

            ticketHtml += `<hr class="my-2 border-stone-200">`;
            ticketHtml += `<div class="flex justify-between"><span>Subtotal:</span> <span>$${subtotal.toFixed(2)} MXN</span></div>`;
            ticketHtml += `<div class="flex justify-between"><span>IVA (16%):</span> <span>$${iva.toFixed(2)} MXN</span></div>`;

            ticketHtml += `<div class="flex justify-between font-bold text-base mt-2 text-purple-900"><span>TOTAL PAGADO:</span> <span>$${finalTotal.toFixed(2)} MXN</span></div>`;

            document.getElementById('ticket-date').innerText = `Emitido el ${new Date().toLocaleDateString()}`;
            document.getElementById('ticket-content').innerHTML = ticketHtml;
            
            document.getElementById('ticket-section').classList.remove('hidden');
            document.getElementById('ticket-section').scrollIntoView({ behavior: 'smooth' });
        }
